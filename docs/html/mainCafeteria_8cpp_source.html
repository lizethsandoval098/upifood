87// Acorta un texto largo para que no se salga de su columna
88// (cuidando no partir a la mitad una letra con acento).
89static string recortar(const string& s, size_t maximo) {
90 if (s.size() <= maximo) return s;
91
92 size_t corte = maximo;
93 while (corte > 0 && (static_cast<unsigned char>(s[corte]) & 0xC0) == 0x80) {
94 corte--;
95 }
96
97 return s.substr(0, corte) + "...";
98}

107// Convierte el texto del campo en una cantidad valida (1 a 9999).
108// Regresa -1 si esta vacio o no sirve.
109static int leerCantidad(const string& texto) {
110 if (texto.empty() || texto.size() > 4) return -1;
111
112 int valor = 0;
113 for (char c : texto) {
114 if (c < '0' || c > '9') return -1;
115 valor = valor * 10 + (c - '0');
116 }
117
118 return (valor >= 1) ? valor : -1;
119}

121// Cantidad de existencias: 0 a 99999. Regresa -1 si no sirve.
122static int leerExistencias(const string& texto) {
123 if (texto.empty() || texto.size() > 5) return -1;
124
125 int valor = 0;
126 for (char c : texto) {
127 if (c < '0' || c > '9') return -1;
128 valor = valor * 10 + (c - '0');
129 }
130
131 return valor;

134// Precio: "12", "12.5" o "12.50". Regresa -1 si no sirve (o si es 0).
135static float leerPrecio(const string& texto) {
136 if (texto.empty() || texto.size() > 8) return -1.0f;
137
138 bool hayPunto = false;
139 bool hayDigito = false;
140
141 for (char c : texto) {
142 if (c == '.') {
143 if (hayPunto) return -1.0f;
144 hayPunto = true;
145 } else if (c >= '0' && c <= '9') {
146 hayDigito = true;
147 } else {
148 return -1.0f;
149 }
150 }
151
152 if (!hayDigito) return -1.0f;
153
154 float valor = 0.0f;
155 try {
156 valor = stof(texto);
157 } catch (...) {
158 return -1.0f;
159 }
160
161 return (valor > 0.0f) ? valor : -1.0f;
162}
