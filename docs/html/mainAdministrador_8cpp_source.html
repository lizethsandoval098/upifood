1// =====================================================================
2// UPIIFOOD - Administrador (interfaz grafica con SFML)
3//
4// IDEA CLAVE (esto es lo que contesta "como se hace lo de SFML junto con
5// todo lo demas"):
6//
7// La VENTANA ES la interfaz. Ya no hay menu en la consola: todo se hace
8// con clics y teclado dentro de la ventana, en UN SOLO CICLO que se
9// repite ~60 veces por segundo:
10//
11// mientras la ventana este abierta:
12// 1) atender eventos (clics, teclas, cerrar) <- pollEvent
13// 2) actualizar datos (pedirle cosas al servidor cuando toque)
14// 3) dibujar todo (clear -> draw -> display)
15//
16// NUNCA hay que quedarse bloqueado esperando (cin, sleep largos...)
17// con la ventana abierta, porque entonces no se atienden eventos y
18// el sistema dice "no responde -> Forzar salida".
19//
20// Pantallas: LOGIN (usuario/contrasena) -> PANEL (botones + tablas)
21// =====================================================================

57// Acorta un texto largo para que no se salga de su columna
58// (cuidando no partir a la mitad una letra con acento).
59static string recortar(const string& s, size_t maximo) {
60 if (s.size() <= maximo) return s;
61
62 size_t corte = maximo;
63 while (corte > 0 && (static_cast<unsigned char>(s[corte]) & 0xC0) == 0x80) {
64 corte--;
65 }
66
67 return s.substr(0, corte) + "...";
68}

71static void rectangulo(Ventana& v, float x, float y, float ancho, float alto, sf::Color color) {
72 sf::RectangleShape r(sf::Vector2f(ancho, alto));

78// Dibuja los titulos de las columnas y la linea de abajo.
79static void encabezadoTabla(Ventana& v, const string& titulo, const string& subtitulo,
80 const vector<pair<string, float>>& columnas) {
81 v.dibujarTexto(titulo, 260, 30, 26, TEXTO_OSCURO);
82 v.dibujarTexto(subtitulo, 260, 66, 15, TEXTO_SUAVE);
83
84 for (const auto& c : columnas) {
85 v.dibujarTexto(c.first, c.second, 104, 16, TEXTO_SUAVE);
86 }
87
88 rectangulo(v, 250, 130, 620, 2, sf::Color(200, 190, 178));

91// Franja de fondo de la fila i (alternando color para que se lea mejor).
92static void franja(Ventana& v, int i) {
93 float y = TABLA_Y_FILAS + i * ALTO_FILA;
94 rectangulo(v, 250, y - 3, 620, ALTO_FILA - 2, (i % 2 == 0) ? FILA_CLARA : FILA_OSCURA);
95}

168 // ---------- 1) EVENTOS ----------
169 sf::Event e;
170 while (win.pollEvent(e)) {
171
172 if (e.type == sf::Event::Closed) {
173 ventana.cerrar();
174 break;
175 }
176
177 if (pantalla == Pantalla::LOGIN) {
178
179 if (e.type == sf::Event::MouseMoved) {
180 botonEntrar.actualizarHover(static_cast<float>(e.mouseMove.x),
181 static_cast<float>(e.mouseMove.y));
182 }
183 else if (e.type == sf::Event::MouseButtonPressed && e.mouseButton.button == sf::Mouse::Left) {
184 float mx = static_cast<float>(e.mouseButton.x);
185 float my = static_cast<float>(e.mouseButton.y);
186
187 // Clic en un campo = darle el foco (el otro lo pierde)
188 campoUsuario.setEnfocado(campoUsuario.contiene(mx, my));
189 campoClave.setEnfocado(campoClave.contiene(mx, my));
190
191 if (botonEntrar.contiene(mx, my)) {
192 intentarLogin();
193 }
194 }
195 else if (e.type == sf::Event::TextEntered) {
196 // Solo el campo que tiene el foco hace caso
197 campoUsuario.recibirCaracter(e.text.unicode);
198 campoClave.recibirCaracter(e.text.unicode);
199 }
200 else if (e.type == sf::Event::KeyPressed) {
201 if (e.key.code == sf::Keyboard::Tab) {
202 bool enUsuario = campoUsuario.estaEnfocado();
203 campoUsuario.setEnfocado(!enUsuario);
204 campoClave.setEnfocado(enUsuario);
205 }
206 else if (e.key.code == sf::Keyboard::Return) {
207 intentarLogin();
208 }
209 }
210 }
211 else { // Pantalla::PANEL
212
213 if (e.type == sf::Event::MouseMoved) {
214 float mx = static_cast<float>(e.mouseMove.x);
215 float my = static_cast<float>(e.mouseMove.y);
216
217 botonUsuarios.actualizarHover(mx, my);
218 botonCafeterias.actualizarHover(mx, my);
219 botonEnLinea.actualizarHover(mx, my);
220 botonSalir.actualizarHover(mx, my);
221 }
222 else if (e.type == sf::Event::MouseButtonPressed && e.mouseButton.button == sf::Mouse::Left) {
223 float mx = static_cast<float>(e.mouseButton.x);
224 float my = static_cast<float>(e.mouseButton.y);
225
226 if (botonUsuarios.contiene(mx, my)) cambiarVista(Vista::USUARIOS);
227 else if (botonCafeterias.contiene(mx, my)) cambiarVista(Vista::CAFETERIAS);
228 else if (botonEnLinea.contiene(mx, my)) cambiarVista(Vista::EN_LINEA);
229 else if (botonSalir.contiene(mx, my)) ventana.cerrar();
230 }
231 else if (e.type == sf::Event::MouseWheelScrolled) {
232 // rueda hacia arriba (delta > 0) = subir en la tabla
233 scroll -= static_cast<int>(e.mouseWheelScroll.delta);
234 scroll = max(0, min(scroll, scrollMaximo));
235 }
236 }
237 }
